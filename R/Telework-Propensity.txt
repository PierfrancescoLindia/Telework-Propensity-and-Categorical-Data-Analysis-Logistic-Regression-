# Definizione della funzione compute_table
compute_table <- function(data_column) {
  data_frame <- data_column %>%
    table() %>%
    as.data.frame() %>%
    rename(Categoria = 1, Frequenza = 2) %>%
    mutate(
      `Percentuale (%)` = Frequenza / sum(Frequenza) * 100,
      `Percentuale Cumulativa (%)` = cumsum(`Percentuale (%)`)
    )
  return(data_frame)
}

# Ricodifica della variabile 'votovi' in binario includendo i NA in "No"
individui <- individui %>%
  mutate(
    votovi_bin = case_when(
      votovi >= 6 ~ "Si",
      TRUE ~ "No"  # Includi tutti gli altri casi (compresi NA) in "No"
    )
  )

# Calcola le frequenze e le percentuali per la variabile ricodificata
votovi_table <- compute_table(individui$votovi_bin)



# Funzione per stampare le tabelle formattate
print_table <- function(data_table, caption) {
  data_table <- data_table %>%
    mutate(
      Frequenza = format(Frequenza, big.mark = ","),
      `Percentuale (%)` = sprintf("%.2f", `Percentuale (%)`),
      `Percentuale Cumulativa (%)` = sprintf("%.2f", `Percentuale Cumulativa (%)`)
    )
  
  kable(data_table, format = "html", col.names = c("Categoria", "Frequenza", "Percentuale (%)", "Percentuale Cumulativa (%)"), caption = caption) %>%
    kable_styling(bootstrap_options = c("striped", "hover", "condensed"), full_width = FALSE, font_size = 12) %>%
    column_spec(1, bold = TRUE, border_right = TRUE)
}

# Creazione della variabile binaria: 1 se "Vorrebbe", 0 altrimenti
individui <- individui %>%
  mutate(telela_bin = ifelse(telela == "Vorrebbe", 1, 0))

# Seleziona solo le variabili demografiche da usare per il modello
dati_model_bin <- individui %>%
  dplyr::select(telela_bin, fascia_eta, fascia_istruzione, sesso, regione, fascia_cittadinanza) %>%
  filter(!is.na(telela_bin))  # Rimuove NA in telela_bin

# Stima del modello logit binario con sole variabili demografiche
modello_logit_bin_demografico <- glm(telela_bin ~ fascia_eta + fascia_istruzione + sesso + regione + fascia_cittadinanza,
                                     family = binomial(link = "logit"),
                                     data = dati_model_bin)

# Sommario del modello
summary(modello_logit_bin_demografico)

# Coefficienti come odds ratio
exp(coef(modello_logit_bin_demografico))

# Predizioni: probabilità di appartenere alla classe "Vorrebbe" (1)
predizioni_bin <- predict(modello_logit_bin_demografico, type = "response")

# Classe predetta (probabilità maggiore di 0.5 significa "Vorrebbe", altrimenti "Non Vorrebbe")
classe_predetta_bin <- ifelse(predizioni_bin > 0.5, 1, 0)

# Matrice di confusione
tabella_confusione <- table(Predetto = classe_predetta_bin, Reale = dati_model_bin$telela_bin)
print(tabella_confusione)

    # Ricodifica della variabile di soddisfazione del tempo libero (esempio)
    sodqnt_bin = case_when(
      sodqnt %in% c("3", "4", "") ~ "Si",  # Includi anche i valori vuoti come neutri
      sodqnt %in% c("1", "2") ~ "No",
      TRUE ~ "Si"  # Assumi che tutti gli altri casi (inclusi NA) siano neutri
    ),
    
    # Ricodifica della variabile di soddisfazione della vita
    votovi_bin = case_when(
      votovi >= 6 ~ "Si",  # Soddisfatto con la vita
      TRUE ~ "No"  # Insoddisfatto con la vita
    ),
    
    # Ricodifica della variabile di soddisfazione lavoro/tempo libero
    sodequ_bin = case_when(
      sodequ %in% c("3", "4", "5") ~ "Si",  # Soddisfatto dell'equilibrio lavoro-tempo libero
      TRUE ~ "No"  # Insoddisfatto
    ),
    
    # Ricodifica della variabile di soddisfazione lavoro
    sodla_bin = case_when(
      sodla %in% c("4", "5") ~ "Si",  # Soddisfatto del lavoro
      TRUE ~ "No"  # Insoddisfatto del lavoro
    ),
    
    # Ricodifica della variabile di soddisfazione finanziaria
    risec_bin = case_when(
      risec %in% c("3", "4") ~ "Si",  # Soddisfatto della situazione finanziaria
      TRUE ~ "No"  # Insoddisfatto
    )
  )

# Seleziona le variabili demografiche e di soddisfazione per il modello
dati_model_soddisfazione <- individui %>%
  dplyr::select(telela_bin, fascia_eta, fascia_istruzione, sesso, regione, fascia_cittadinanza, sodqnt_bin, votovi_bin, sodequ_bin, sodla_bin, risec_bin) %>%
  filter(!is.na(telela_bin))  # Rimuove NA in telela_bin

# ----------------------------------------------------------
# Stima del modello logit binario con variabili di soddisfazione
# ----------------------------------------------------------

modello_logit_soddisfazione <- glm(telela_bin ~ fascia_eta + fascia_istruzione + sesso + regione + fascia_cittadinanza + sodqnt_bin + votovi_bin + sodequ_bin + sodla_bin + risec_bin,
                                   family = binomial(link = "logit"),
                                   data = dati_model_soddisfazione)

# Sommario del modello
summary(modello_logit_soddisfazione)

# Coefficienti come odds ratio
exp(coef(modello_logit_soddisfazione))

################################################################################
############################### CONFRONTO TRA MODELLI #########################
# Confronta AIC dei due modelli
aic_modello_demografico <- AIC(modello_logit_demografico)
aic_modello_soddisfazione <- AIC(modello_logit_soddisfazione)

cat("AIC modello demografico:", aic_modello_demografico, "\n")
cat("AIC modello soddisfazione:", aic_modello_soddisfazione, "\n")




####ANOVA MODELLO DEMO E MODELLO DEM + SODD

# Confronta il modello con variabili demografiche (modello ridotto) e il modello con variabili demografiche + soddisfazione (modello completo)
anova(modello_logit_bin, modello_logit_soddisfazione, test = "Chisq")
